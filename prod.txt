- name: Deploy to server
        uses: appleboy/ssh-action@v0.1.5
        with:
          host: ${{ secrets.SERVER_HOST }}
          username: ${{ secrets.SERVER_USER }}
          key: ${{ secrets.SSH_KEY }}
          script: |
            docker pull ${{ secrets.DOCKER_USERNAME }}/insurance-backend:latest
            docker pull ${{ secrets.DOCKER_USERNAME }}/insurance-frontend:latest
            docker-compose down
            docker-compose up -d
